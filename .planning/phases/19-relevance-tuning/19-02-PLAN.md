         {unrecognized, recognized} = Map.pop(canonical, :__unrecognized__, %{})

         atom
         |> Atom.to_string()
         |> Macro.camelize()
         |> then(fn s -> String.downcase(String.slice(s, 0..0)) <> String.slice(s, 1..-1//1) end)

               is_binary(k) -> k

         end)
